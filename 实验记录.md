### 生成图片

- 通过SD14本地模型，借助prompt集合文件，使用python脚本批量生成对应图片

```python
import os
import torch
from diffusers import StableDiffusionPipeline

# ===== 路径配置 =====
MODEL_PATH = "/root/autodl-tmp/models/stable-diffusion/stable-diffusion-v1-4"
PROMPT_FILE = "/root/autodl-tmp/code/stable-diffusion/sd14/sd14_prompts.txt"
OUTPUT_DIR = "/root/autodl-tmp/outputs/sd14_test"

# ===== 推理参数 =====
HEIGHT = 512
WIDTH = 512
NUM_INFERENCE_STEPS = 30
GUIDANCE_SCALE = 7.5
SEED = 42

def load_prompts(prompt_file: str):
    """
    读取 prompt 文件
    格式：
    文件名<TAB>正向prompt<TAB>negative prompt
    跳过空行和以 # 开头的注释行
    """
    tasks = []
    with open(prompt_file, "r", encoding="utf-8") as f:
        for line_num, line in enumerate(f, start=1):
            line = line.strip()
            if not line or line.startswith("#"):
                continue

            parts = line.split("\t")
            if len(parts) < 2:
                print(f"[Warning] 第 {line_num} 行格式不正确，已跳过: {line}")
                continue

            filename = parts[0].strip()
            prompt = parts[1].strip()
            negative_prompt = parts[2].strip() if len(parts) > 2 else ""

            tasks.append({
                "filename": filename,
                "prompt": prompt,
                "negative_prompt": negative_prompt
            })
    return tasks


def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    print("Loading pipeline...")
    pipe = StableDiffusionPipeline.from_pretrained(
        MODEL_PATH,
        torch_dtype=torch.float16,
        safety_checker=None,
        requires_safety_checker=False,
        local_files_only=True
    )
    pipe = pipe.to("cuda")

    # 显存优化
    pipe.enable_attention_slicing()
    try:
        pipe.enable_xformers_memory_efficient_attention()
        print("xformers enabled.")
    except Exception as e:
        print(f"xformers not enabled: {e}")

    tasks = load_prompts(PROMPT_FILE)
    print(f"Loaded {len(tasks)} prompts.")

    for idx, task in enumerate(tasks, start=1):
        filename = task["filename"]
        prompt = task["prompt"]
        negative_prompt = task["negative_prompt"]

        print(f"[{idx}/{len(tasks)}] Generating: {filename}")
        print(f"  prompt: {prompt}")
        if negative_prompt:
            print(f"  negative: {negative_prompt}")

        generator = torch.Generator(device="cuda").manual_seed(SEED)

        image = pipe(
            prompt=prompt,
            negative_prompt=negative_prompt,
            num_inference_steps=NUM_INFERENCE_STEPS,
            guidance_scale=GUIDANCE_SCALE,
            generator=generator,
            height=HEIGHT,
            width=WIDTH
        ).images[0]

        save_path = os.path.join(OUTPUT_DIR, f"{filename}.png")
        image.save(save_path)
        print(f"  saved to: {save_path}")

    print("All images generated successfully.")


if __name__ == "__main__":
    main()
```

#### 测试训练效果

- 编写推理脚本，使用本地sd14模型，加载LoRA权重文件，使用较为简单的测试prompt，生成图片
  - 结果保存在`/root/autodl-tmp/outputs/lora_cat_statue/samples中`
- **从图片中可以看到，模型从6张样本中学习到了cat_statue**

```bash
import os
import torch
from diffusers import StableDiffusionPipeline

# ===== 路径配置 =====
BASE_MODEL_PATH = "/root/autodl-tmp/models/stable-diffusion/stable-diffusion-v1-4"
LORA_PATH = "/root/autodl-tmp/outputs/lora_cat_statue/checkpoints"
OUTPUT_DIR = "/root/autodl-tmp/outputs/lora_cat_statue/samples"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# ===== 测试 prompt =====
TEST_PROMPTS = [
    "a photo of vks cat_statue",
    "a close-up photo of vks cat_statue",
    "a vks cat_statue on a table",
    "a vks cat_statue in the snow",
]

NEGATIVE_PROMPT = "blurry, low quality, distorted, realistic cat, furry cat, real animal"

# ===== 推理参数 =====
NUM_INFERENCE_STEPS = 30
GUIDANCE_SCALE = 7.5
HEIGHT = 512
WIDTH = 512
SEED = 42


def main():
    print("Loading base model...")
    pipe = StableDiffusionPipeline.from_pretrained(
        BASE_MODEL_PATH,
        torch_dtype=torch.float16,
        safety_checker=None,
        requires_safety_checker=False,
        local_files_only=True,
    ).to("cuda")

    pipe.enable_attention_slicing()

    print(f"Loading LoRA from: {LORA_PATH}")
    pipe.load_lora_weights(LORA_PATH)

    for i, prompt in enumerate(TEST_PROMPTS, start=1):
        generator = torch.Generator(device="cuda").manual_seed(SEED)

        print(f"[{i}/{len(TEST_PROMPTS)}] Generating: {prompt}")
        image = pipe(
            prompt=prompt,
            negative_prompt=NEGATIVE_PROMPT,
            num_inference_steps=NUM_INFERENCE_STEPS,
            guidance_scale=GUIDANCE_SCALE,
            height=HEIGHT,
            width=WIDTH,
            generator=generator,
        ).images[0]

        save_path = os.path.join(OUTPUT_DIR, f"sample_{i}.png")
        image.save(save_path)
        print(f"Saved to: {save_path}")

    print("Done.")


if __name__ == "__main__":
    main()
```
